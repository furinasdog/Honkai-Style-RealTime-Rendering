cmake_minimum_required(VERSION 3.20)
project(archive_service_tests LANGUAGES CXX)
add_executable(archive_service_tests archive_service_tests.cpp ../../src/archive_service.cpp)
target_compile_features(archive_service_tests PRIVATE cxx_std_17)
target_include_directories(archive_service_tests PRIVATE ../../src ../../third_party/7zip/CPP)

endif()
