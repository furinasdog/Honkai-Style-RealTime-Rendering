cmake_minimum_required(VERSION 3.20)
project(mocap_native LANGUAGES CXX)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/.gdignore" "")
set(CMAKE_CXX_STANDARD 17)
set(GODOT_EXECUTABLE "" CACHE FILEPATH "Godot 4.7 Windows editor executable, used only to export its native API")
if(NOT EXISTS "${GODOT_EXECUTABLE}")
    message(FATAL_ERROR "Set GODOT_EXECUTABLE to a Godot 4.7 editor executable")
endif()
set(API_DIR "${CMAKE_CURRENT_BINARY_DIR}/godot_api")
file(MAKE_DIRECTORY "${API_DIR}")
file(WRITE "${API_DIR}/project.godot" "config_version=5\n")
execute_process(COMMAND "${GODOT_EXECUTABLE}" --headless --path "${API_DIR}" --dump-gdextension-interface --dump-extension-api
    WORKING_DIRECTORY "${API_DIR}" RESULT_VARIABLE API_RESULT OUTPUT_QUIET ERROR_VARIABLE API_ERROR)
if(NOT API_RESULT EQUAL 0)
    message(FATAL_ERROR "Godot native API generation failed: ${API_ERROR}")
endif()
file(READ "${API_DIR}/extension_api.json" GODOT_API)
string(JSON CONFIG_COUNT LENGTH "${GODOT_API}" builtin_class_sizes)
math(EXPR CONFIG_LAST "${CONFIG_COUNT} - 1")
foreach(CONFIG_INDEX RANGE ${CONFIG_LAST})
    string(JSON CONFIG_NAME GET "${GODOT_API}" builtin_class_sizes ${CONFIG_INDEX} build_configuration)
    if(CONFIG_NAME STREQUAL "float_64")
        string(JSON SIZE_COUNT LENGTH "${GODOT_API}" builtin_class_sizes ${CONFIG_INDEX} sizes)
        math(EXPR SIZE_LAST "${SIZE_COUNT} - 1")
        foreach(SIZE_INDEX RANGE ${SIZE_LAST})
            string(JSON TYPE_NAME GET "${GODOT_API}" builtin_class_sizes ${CONFIG_INDEX} sizes ${SIZE_INDEX} name)
            string(JSON TYPE_SIZE GET "${GODOT_API}" builtin_class_sizes ${CONFIG_INDEX} sizes ${SIZE_INDEX} size)
            set(SIZE_${TYPE_NAME} ${TYPE_SIZE})
        endforeach()
    endif()
endforeach()
option(BUILD_NATIVE_TESTS "Build native validation executables (not distributed)" OFF)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../plugins/honkai_mmd" "${CMAKE_CURRENT_BINARY_DIR}/honkai_mmd")
if(BUILD_NATIVE_TESTS)
    add_executable(pmx_parser_test tests/pmx_parser_test.cpp)
    target_link_libraries(pmx_parser_test PRIVATE pmx_parser)
    enable_testing()
    add_test(NAME pmx_parser_validation COMMAND pmx_parser_test)
    if(MSVC)
        target_compile_options(pmx_parser_test PRIVATE /utf-8)
        set_property(TARGET pmx_parser_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    add_executable(spout_receiver tests/spout_receiver.cpp)
    target_link_libraries(spout_receiver PRIVATE spout_native ole32 oleaut32 uuid)
    if(MSVC)
        set_property(TARGET spout_receiver PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
endif()
if(NOT SIZE_Variant OR NOT SIZE_PackedFloat32Array)
    message(FATAL_ERROR "Godot API does not expose the expected Windows x64 single-precision ABI")
endif()
configure_file(src/api_sizes.hpp.in "${API_DIR}/api_sizes.hpp" @ONLY)
set(SPOUT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/Spout2/SPOUTSDK")
add_library(spout_native STATIC
    "${SPOUT_DIR}/SpoutDirectX/SpoutDX/SpoutDX.cpp"
    "${SPOUT_DIR}/SpoutGL/SpoutCopy.cpp" "${SPOUT_DIR}/SpoutGL/SpoutDirectX.cpp"
    "${SPOUT_DIR}/SpoutGL/SpoutFrameCount.cpp" "${SPOUT_DIR}/SpoutGL/SpoutSenderNames.cpp"
    "${SPOUT_DIR}/SpoutGL/SpoutSharedMemory.cpp" "${SPOUT_DIR}/SpoutGL/SpoutUtils.cpp")
target_compile_definitions(spout_native PUBLIC SPOUT_BUILD_STATIC PRIVATE _CRT_SECURE_NO_WARNINGS)
target_include_directories(spout_native PUBLIC "${SPOUT_DIR}/SpoutDirectX/SpoutDX" "${SPOUT_DIR}/SpoutGL")
target_link_libraries(spout_native PUBLIC d3d11 dxgi user32 gdi32 comctl32 shell32 advapi32 winmm version psapi)
add_library(mocap_native SHARED src/godot_bridge.cpp src/mediapipe_core.cpp src/windows_camera.cpp src/archive_service.cpp src/spout_output.cpp)
target_include_directories(mocap_native PRIVATE third_party/mediapipe/include third_party/7zip/CPP third_party/7zip/C "${API_DIR}")
target_compile_definitions(mocap_native PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(mocap_native PRIVATE mf mfplat mfreadwrite mfuuid ole32 oleaut32 bcrypt normaliz uuid d3d12 spout_native pmx_parser)
if(MSVC)
    target_compile_options(mocap_native PRIVATE /utf-8 /W4)
    set_property(TARGET mocap_native PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    set_property(TARGET spout_native PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set_target_properties(mocap_native PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
foreach(configuration Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER ${configuration} configuration_upper)
    set_target_properties(mocap_native PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${configuration_upper} "${CMAKE_CURRENT_SOURCE_DIR}/bin")
endforeach()
